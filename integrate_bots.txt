function Write-NTRLIStatus {
    param([string]$Message, [string]$Type = "Info")
    
    switch ($Type) {
        "Success" { Write-Host "  ✓ " -ForegroundColor Green -NoNewline; Write-Host $Message }
        "Error" { Write-Host "  ✗ " -ForegroundColor Red -NoNewline; Write-Host $Message }
        "Warning" { Write-Host "  ⚠ " -ForegroundColor Yellow -NoNewline; Write-Host $Message }
        "Info" { Write-Host "  ℹ " -ForegroundColor Cyan -NoNewline; Write-Host $Message }
        "Progress" { Write-Host "  ⟳ " -ForegroundColor Cyan -NoNewline; Write-Host $Message }
    }
}

function Test-Prerequisites {
    Write-NTRLIStatus "Checking prerequisites..." "Progress"
    
    # Check Python
    try {
        $pythonVersion = python --version 2>&1
        Write-NTRLIStatus "Python found: $pythonVersion" "Success"
    }
    catch {
        Write-NTRLIStatus "Python not found! Install from python.org" "Error"
        return $false
    }
    
    # Check pip packages
    $requiredPackages = @("telethon", "python-dotenv", "aiohttp")
    foreach ($package in $requiredPackages) {
        $installed = pip show $package 2>$null
        if ($installed) {
            Write-NTRLIStatus "Package '$package' installed" "Success"
        }
        else {
            Write-NTRLIStatus "Installing '$package'..." "Progress"
            pip install $package --quiet
            Write-NTRLIStatus "Package '$package' installed" "Success"
        }
    }
    
    return $true
}

function Update-BotScriptWithHealing {
    param([string]$ScriptPath)
    
    $scriptName = Split-Path $ScriptPath -Leaf
    Write-NTRLIStatus "Updating $scriptName with self-healing..." "Progress"
    
    $content = Get-Content $ScriptPath -Raw
    
    # Check if already integrated
    if ($content -match "from self_heal import get_monitor") {
        Write-NTRLIStatus "$scriptName already has self-healing" "Info"
        return
    }
    
    # Add self-healing import after other imports
    $healingImport = @"

# Self-Healing Integration
from self_heal import get_monitor
health_monitor = get_monitor()
"@
    
    # Find the right place to insert (after imports, before main code)
    if ($content -match "(?s)(import.*?\n\n)") {
        $content = $content -replace "(?s)(import.*?\n\n)", "`$1$healingImport`n"
    }
    
    # Add health monitoring to main/start function
    $monitorStart = @"

    # Start health monitor
    try:
        health_monitor.start()
        logger.info("✓ Self-healing monitor active")
    except Exception as e:
        logger.warning(f"Health monitor could not start: {e}")
"@
    
    # Insert monitor start in main/start_bot function
    if ($content -match "(?s)(async def (main|start_bot)\(\):.*?try:)") {
        $content = $content -replace "(?s)(async def (main|start_bot)\(\):.*?try:)", "`$1$monitorStart"
    }
    
    # Save updated content
    Set-Content -Path $ScriptPath -Value $content
    Write-NTRLIStatus "$scriptName updated successfully" "Success"
}

function Start-HealthCheck {
    Write-NTRLIStatus "Running health diagnostics..." "Progress"
    Write-Host ""
    
    # Check data integrity
    $dataPath = Join-Path $ProjectRoot "data"
    $dataFiles = Get-ChildItem -Path $dataPath -Filter "*.json" -ErrorAction SilentlyContinue
    
    foreach ($file in $dataFiles) {
        try {
            $json = Get-Content $file.FullName | ConvertFrom-Json
            Write-NTRLIStatus "Data file OK: $($file.Name)" "Success"
        }
        catch {
            Write-NTRLIStatus "Data file corrupted: $($file.Name)" "Error"
        }
    }
    
    # Check logs
    $logPath = Join-Path $ProjectRoot "logs"
    if (Test-Path $logPath) {
        $logFiles = Get-ChildItem -Path $logPath -Filter "*.log"
        $totalSize = ($logFiles | Measure-Object -Property Length -Sum).Sum / 1MB
        Write-NTRLIStatus "Total log size: $([math]::Round($totalSize, 2)) MB" "Info"
        
        if ($totalSize -gt 50) {
            Write-NTRLIStatus "Log files are large, consider archiving" "Warning"
        }
    }
    
    # Check health status
    $healthFile = Join-Path $ProjectRoot "data/health.json"
    if (Test-Path $healthFile) {
        $health = Get-Content $healthFile | ConvertFrom-Json
        Write-Host ""
        Write-Host "  Last Health Check:" -ForegroundColor Cyan
        Write-Host "    Status: $($health.status)" -ForegroundColor $(if ($health.status -eq "healthy") { "Green" } else { "Yellow" })
        Write-Host "    Timestamp: $($health.timestamp)"
        Write-Host "    Errors: $($health.errors)"
        Write-Host "    Restarts: $($health.restarts)"
    }
    
    Write-Host ""
}
